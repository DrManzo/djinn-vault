```bash

[Unit]
Description=Typhon heartbeat writer
[Service]
Type=oneshot
ExecStart=/bin/bash -c '
  CPU_TEMP=$(cat /sys/class/thermal/thermal_zone0/temp 2>/dev/null)
  CPU_C=$((CPU_TEMP / 1000))
  GPU_TEMP=$(nvidia-smi --query-gpu=temperature.gpu --format=csv,noheader 2>/dev/null || echo "N/A")
  RAM_USED=$(free -h | awk "/Mem:/ {print \$3}")
  RAM_TOTAL=$(free -h | awk "/Mem:/ {print \$2}")
  DISK=$(df -h ~/Obsidian | tail -1 | awk "{print \$4}")
  cat > ~/Obsidian/djinn/communications/TYPHON-HEARTBEAT.md << BEAT
# Typhon Heartbeat
Updated: $(date)
CPU: ${CPU_C}°C
GPU: ${GPU_TEMP}°C
RAM: ${RAM_USED} / ${RAM_TOTAL}
Disk free: ${DISK}
Ollama: $(systemctl is-active ollama)
BEAT
  cd ~/Obsidian && git add djinn/communications/TYPHON-HEARTBEAT.md
  git commit -m "heartbeat: typhon $(date +%H:%M)" && git push'
EOF

[Unit]
Description=Typhon heartbeat every 5 min
[Timer]
OnBootSec=1min
OnUnitActiveSec=5min
Persistent=true
[Install]
WantedBy=timers.target
EOF
```
