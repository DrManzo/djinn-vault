Add Hunyuan3D wrapper to Djinn toolkit — `~/.local/bin/djinn-generate-3d`:

```bash
#!/bin/bash
# Usage: djinn-generate-3d "a wall bracket with two screw holes"
PROMPT="$1"
OUTPUT_DIR="$HOME/Obsidian/djinn/printer/queue"
TIMESTAMP=$(date +%Y%m%d_%H%M%S)
OUTPUT_FILE="$OUTPUT_DIR/generated_${TIMESTAMP}.glb"

source ~/.venvs/hunyuan3d/bin/activate
python3 - << EOF
from hy3dgen.texgen import Hunyuan3DDiTFlowMatchingPipeline
pipe = Hunyuan3DDiTFlowMatchingPipeline.from_pretrained('tencent/Hunyuan3D-2')
mesh = pipe("$PROMPT")[0]
mesh.export("$OUTPUT_FILE")
print(f"Generated: $OUTPUT_FILE")
EOF

# Drop vault note
cat > "$OUTPUT_DIR/../active/generated_${TIMESTAMP}.md" << VAULTNOTE
---
job_id: generated_${TIMESTAMP}
file: generated_${TIMESTAMP}.glb
status: queued
```

```python
# /print command handler
async def handle_print(update, context):
    args = context.args
    if not args:
        await update.message.reply_text("Usage: /print <filename.stl>")
        return

    filename = args[0]
    stl_path = f"{VAULT_PATH}/djinn/printer/queue/{filename}"

    if not os.path.exists(stl_path):
        await update.message.reply_text(f"File not found in queue: {filename}")
        return

    # Send to Moonraker
    async with aiohttp.ClientSession() as session:
        resp = await session.post(
            f"{MOONRAKER_URL}/printer/print/start",
            json={"filename": filename}
        )
        data = await resp.json()

    if data.get("result") == "ok":
        await update.message.reply_text(f"🖨️ Print started: {filename}")
    else:
        await update.message.reply_text(f"❌ Print failed: {data}")

# /print_status command handler
async def handle_print_status(update, context):
    async with aiohttp.ClientSession() as session:
        resp = await session.get(f"{MOONRAKER_URL}/printer/objects/query?print_stats")
        data = await resp.json()

    stats = data["result"]["status"]["print_stats"]
    state = stats.get("state", "unknown")
    filename = stats.get("filename", "none")
    progress = stats.get("print_duration", 0)

    await update.message.reply_text(
        f"🖨️ Printer status\n"
        f"State: {state}\n"
        f"File: {filename}\n"
        f"Duration: {int(progress//60)}m"
    )
```
